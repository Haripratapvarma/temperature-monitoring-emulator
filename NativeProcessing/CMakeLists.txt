cmake_minimum_required(VERSION 3.20)
project(templab_native VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

# Put the shared library where the managed projects look for it (build/bin), for every config.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
foreach(cfg Debug Release RelWithDebInfo MinSizeRel)
  string(TOUPPER ${cfg} CFG)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${CFG} ${CMAKE_BINARY_DIR}/bin)
  set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${CFG} ${CMAKE_BINARY_DIR}/bin)
endforeach()

if(MSVC)
  set(TL_WARNINGS /W4 /permissive-)
else()
  set(TL_WARNINGS -Wall -Wextra -Wpedantic -Werror -Wconversion -Wno-sign-conversion)
endif()

add_library(templab_native SHARED src/rolling_stats_c.cpp)
target_include_directories(templab_native PUBLIC include PRIVATE src)
target_compile_definitions(templab_native PRIVATE TEMPLAB_NATIVE_BUILD)
target_compile_options(templab_native PRIVATE ${TL_WARNINGS})
set_target_properties(templab_native PROPERTIES PREFIX "lib")
if(WIN32)
  set_target_properties(templab_native PROPERTIES PREFIX "")
endif()

option(TL_BUILD_TESTS "Build native tests" ON)
option(TL_SANITIZE "Build tests with address/undefined sanitizers (GCC/Clang)" OFF)

if(TL_BUILD_TESTS)
  enable_testing()
  add_executable(test_rolling_stats tests/test_rolling_stats.cpp)
  target_link_libraries(test_rolling_stats PRIVATE templab_native)
  add_test(NAME rolling_stats COMMAND test_rolling_stats)

  add_executable(bench_rolling_stats bench/bench_rolling_stats.cpp)
  target_link_libraries(bench_rolling_stats PRIVATE templab_native)

  if(TL_SANITIZE AND NOT MSVC)
    foreach(t templab_native test_rolling_stats)
      target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${t} PRIVATE -fsanitize=address,undefined)
    endforeach()
  endif()
endif()
